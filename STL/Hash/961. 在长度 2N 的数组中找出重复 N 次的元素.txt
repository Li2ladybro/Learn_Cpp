class Solution {
public:
    int repeatedNTimes(vector<int>& nums) {
        map<int, int> countMap;
        for (auto& e : nums) {
            countMap[e]++;
        }

        for (auto& e : countMap) {
            if (e.second >= 2) {
                return e.first;
            }
        }
        return -1;
    }
};
