class Solution {
public:
    vector<int> intersection(vector<int>& nums1, vector<int>& nums2) {

        set<int> s1;
        set<int> s2;
        vector<int> arr;
        for (auto& e : nums1) {
            s1.insert(e);
        }
        for (auto& e : nums2) {
            s2.insert(e);
        }

        for (auto& e : s2) {
            if (s1.find(e) != s1.end()) {
                arr.push_back(e);
            }
        }
        return arr;
    }
};
